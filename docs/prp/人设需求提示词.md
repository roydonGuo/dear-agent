# 角色
你现在是一位大模型应用开发工程师，熟练spring ai和java生态，现在请你先思考，再完成一下需求：
# 需求
现有两张数据表：
-- 1. 人设分类数据表 ai_prompt_category
CREATE TABLE `ai_prompt_category` (
`id` bigint NOT NULL AUTO_INCREMENT COMMENT '主键ID',
`name` varchar(50) NOT NULL COMMENT '分类名',
`sort` int NOT NULL DEFAULT '0' COMMENT '排序优先级',
`icon` varchar(20) DEFAULT NULL COMMENT '分类图标名称（lucide 图标类名）',
`create_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
`update_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='人设分类数据表';

-- 2. 人设表 ai_prompt
CREATE TABLE `ai_prompt` (
`id` bigint NOT NULL AUTO_INCREMENT COMMENT '主键ID',
`title` varchar(50) NOT NULL COMMENT '人设标题',
`avatar` varchar(100) DEFAULT NULL COMMENT '人设图标（图片链接）',
`description` varchar(500) DEFAULT NULL COMMENT '人设描述',
`prompt` longtext NOT NULL COMMENT '人设提示词',
`category_ids` longtext DEFAULT NULL COMMENT '人设分类 ids，逗号分隔的分类ID字符串',
`create_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
`update_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='人设表';
一个人设分类下有多个人设，一个人设可以有多个分类（通过category_ids关联，可以用逗号分隔的分类ID字符串关联）。
现在你要完成对这两个表的增删查改和列表查询，人设分类是查询集合，人设是分页查询且可传递人设分类IDS参数。
